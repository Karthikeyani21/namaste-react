# React context
  - can able to use the data across application wide

  # createContext
    - can create context using createContext from react
    const UserContext = createContext({
        loggedInUser: "Karthi"
    })

    // access context in react
    # useContext hook
      - react hook, used to use the created context from the application
      useContext(UserContext)
    # class based component
      - <UserContext.Consumer>
           {(data)=> { context data.....}}
        </UserContext.Consumer>

    # modify the context
       - <UserContext.Provider value= {{loggedInUser: "new value"}}>
              // component which will use the updated context should be wrapped inside the provider
              // we can pass setState to the value, so that we can modify the context value
               {loggedInUser: "new value", setUserName}

        </UserContext.Provider>
